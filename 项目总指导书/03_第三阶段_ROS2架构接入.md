# 第三阶段：ROS2 架构接入（第 9-12 周）

## 本阶段目标

将 C++ 控制器封装为 ROS2 节点，通过 ROS2 话题与 CARLA 通信，构建完整的发布-订阅架构。

**里程碑：全程通过 ROS2 话题通信，CARLA 中车辆跟踪轨迹，用 `ros2 topic echo` 能看到实时控制指令。**

---

## 为什么要接入 ROS2

真实自动驾驶系统中，感知/规划/控制是独立进程，通过消息中间件通信。ROS2 是行业标准。

```
没有 ROS2（第二阶段）：
  Python CARLA 脚本 ←管道→ C++ 控制器（单进程，耦合紧）

有 ROS2（第三阶段）：
  轨迹发布节点 → /trajectory 话题 → 控制器节点 → /vehicle_cmd 话题 → CARLA 桥接节点
  （各节点独立进程，解耦，可替换任意模块）
```

这一架构与 Autoware 的设计思路一致：控制器封装为 ROS2 节点，通过话题订阅规划模块的轨迹，发布控制指令。

### 9.3 ROS2 核心概念速览

**节点（Node）：** 一个独立运行的进程，承担特定功能（发轨迹、做控制、驱动 CARLA）

**话题（Topic）：** 节点间传递数据的通道，发布者不关心谁在接收

**消息（Message）：** 话题上传递的数据类型，如 `geometry_msgs/Twist`、自定义消息

**QoS：** 消息传输质量策略（可靠 vs 尽力而为，队列深度等）

```
轨迹发布节点                控制器节点               CARLA 桥接节点
(trajectory_publisher)     (controller_node)       (carla_bridge_node)
        |                          |                       |
        |── /trajectory ──────────>|                       |
        |   AV轨迹点列表            |── /vehicle_cmd ──────>|
        |                          |   油门/刹车/转角        |── CARLA API ──> 车辆
        |<── /vehicle_state ───────────────────────────────|
            x,y,theta,v
```

创建 `av_controller/src/controller_node.cpp`：

```cpp
#include <rclcpp/rclcpp.hpp>
#include <chrono>
#include <memory>
#include <vector>

#include "av_control_msgs/msg/trajectory.hpp"
#include "av_control_msgs/msg/vehicle_state.hpp"
#include "av_control_msgs/msg/vehicle_cmd.hpp"

// 引入第二阶段写好的控制器（复用代码）
#include "pid_controller.hpp"
#include "pure_pursuit.hpp"
#include "stanley_controller.hpp"

using namespace std::chrono_literals;
using Trajectory  = av_control_msgs::msg::Trajectory;
using VehicleState = av_control_msgs::msg::VehicleState;
using VehicleCmd  = av_control_msgs::msg::VehicleCmd;

class ControllerNode : public rclcpp::Node {
public:
    ControllerNode() : Node("controller_node") {
        // 声明参数（可通过命令行或 yaml 配置文件修改）
        this->declare_parameter("controller_type", "pure_pursuit");
        this->declare_parameter("pid_kp", 1.5);
        this->declare_parameter("pid_ki", 0.1);
        this->declare_parameter("pid_kd", 0.05);
        this->declare_parameter("pp_k", 0.1);
        this->declare_parameter("pp_ld_min", 2.0);

        // 读取参数
        std::string ctrl_type = this->get_parameter("controller_type").as_string();
        double kp = this->get_parameter("pid_kp").as_double();
        double ki = this->get_parameter("pid_ki").as_double();
        double kd = this->get_parameter("pid_kd").as_double();

        pid_ = std::make_unique<PIDController>(kp, ki, kd);
        pure_pursuit_ = std::make_unique<PurePursuitController>(
            2.7,
            this->get_parameter("pp_k").as_double(),
            this->get_parameter("pp_ld_min").as_double()
        );

        RCLCPP_INFO(this->get_logger(), "控制器启动: %s", ctrl_type.c_str());

        // 订阅轨迹
        traj_sub_ = this->create_subscription<Trajectory>(
            "/trajectory", 10,
            [this](const Trajectory::SharedPtr msg) {
                trajectory_.clear();
                for (size_t i = 0; i < msg->x.size(); ++i) {
                    trajectory_.push_back({
                        msg->x[i], msg->y[i],
                        msg->yaw[i], msg->v_ref[i]
                    });
                }
                RCLCPP_INFO_ONCE(this->get_logger(),
                    "收到轨迹，共 %zu 个路点", trajectory_.size());
            });

        // 订阅车辆状态
        state_sub_ = this->create_subscription<VehicleState>(
            "/vehicle_state", 10,
            [this](const VehicleState::SharedPtr msg) {
                current_state_ << msg->x, msg->y, msg->theta, msg->v;
                state_received_ = true;
            });

        // 发布控制指令
        cmd_pub_ = this->create_publisher<VehicleCmd>("/vehicle_cmd", 10);

        // 20Hz 控制循环
        timer_ = this->create_wall_timer(50ms, [this]() { controlLoop(); });
    }

private:
    void controlLoop() {
        if (!state_received_ || trajectory_.empty()) return;

        auto [delta, v_ref] = pure_pursuit_->compute(current_state_, trajectory_);
        double accel = pid_->compute(v_ref, current_state_[3]);

        VehicleCmd cmd;
        cmd.header.stamp = this->now();
        cmd.delta  = delta;
        cmd.accel  = accel;
        cmd.throttle = std::max(accel / 3.0, 0.0);
        cmd.brake    = std::max(-accel / 5.0, 0.0);
        cmd.steer    = std::clamp(delta / 0.6, -1.0, 1.0);

        cmd_pub_->publish(cmd);
    }

    // 成员变量
    std::unique_ptr<PIDController>         pid_;
    std::unique_ptr<PurePursuitController> pure_pursuit_;

    std::vector<PurePursuitController::Waypoint> trajectory_;
    Eigen::Vector4d current_state_ = Eigen::Vector4d::Zero();
    bool state_received_ = false;

    rclcpp::Subscription<Trajectory>::SharedPtr   traj_sub_;
    rclcpp::Subscription<VehicleState>::SharedPtr state_sub_;
    rclcpp::Publisher<VehicleCmd>::SharedPtr      cmd_pub_;
    rclcpp::TimerBase::SharedPtr                  timer_;
};

int main(int argc, char* argv[]) {
    rclcpp::init(argc, argv);
    rclcpp::spin(std::make_shared<ControllerNode>());
    rclcpp::shutdown();
    return 0;
}
```

### 11.2 轨迹发布节点

创建 `av_controller/src/trajectory_publisher.cpp`：

```cpp
#include <rclcpp/rclcpp.hpp>
#include <cmath>
#include "av_control_msgs/msg/trajectory.hpp"

using namespace std::chrono_literals;
using Trajectory = av_control_msgs::msg::Trajectory;

class TrajectoryPublisher : public rclcpp::Node {
public:
    TrajectoryPublisher() : Node("trajectory_publisher") {
        pub_ = this->create_publisher<Trajectory>("/trajectory", 10);

        // 生成 S 形轨迹
        traj_msg_ = generateSCurve();

        // 每秒发布一次（轨迹不常变，不需要高频）
        timer_ = this->create_wall_timer(1000ms, [this]() {
            traj_msg_.header.stamp = this->now();
            pub_->publish(traj_msg_);
            RCLCPP_INFO_ONCE(this->get_logger(), "轨迹已发布，共 %zu 个路点",
                             traj_msg_.x.size());
        });
    }

private:
    Trajectory generateSCurve() {
        Trajectory msg;
        const int N = 300;
        const double length = 100.0;

        for (int i = 0; i < N; ++i) {
            double x = length * i / N;
            double y = 10.0 * std::sin(2.0 * M_PI * x / length);
            double dy_dx = 10.0 * std::cos(2.0 * M_PI * x / length)
                           * (2.0 * M_PI / length);
            double yaw = std::atan2(dy_dx, 1.0);

            msg.x.push_back(x);
            msg.y.push_back(y);
            msg.yaw.push_back(yaw);
            msg.v_ref.push_back(6.0);
        }
        return msg;
    }

    rclcpp::Publisher<Trajectory>::SharedPtr pub_;
    rclcpp::TimerBase::SharedPtr timer_;
    Trajectory traj_msg_;
};

int main(int argc, char* argv[]) {
    rclcpp::init(argc, argv);
    rclcpp::spin(std::make_shared<TrajectoryPublisher>());
    rclcpp::shutdown();
    return 0;
}
```

### 11.3 仿真桥接节点（Python）

创建 `av_controller/scripts/sim_bridge_node.py`：

```python
#!/usr/bin/env python3
"""
动态自行车模型 ROS2 桥接节点
- 订阅 /vehicle_cmd，驱动 Python 仿真器步进
- 发布 /vehicle_state（仿真器当前状态）
"""

import sys
import os
sys.path.append(os.path.expanduser('~/av_control_benchmark/python_prototype'))

import rclpy
from rclpy.node import Node
import numpy as np

from av_control_msgs.msg import VehicleCmd, VehicleState
from models.dynamic_bicycle_model import DynamicBicycleModel
from utils.trajectory_generator import generate_s_curve

class SimBridgeNode(Node):
    def __init__(self):
        super().__init__('sim_bridge_node')

        # 初始化仿真器
        self.model = DynamicBicycleModel(dt=0.05)
        traj = generate_s_curve()
        self.model.set_state(traj[0,0], traj[0,1], traj[0,2], 3.0)
        self.get_logger().info('仿真器已初始化，起点: ({:.1f}, {:.1f})'.format(
            traj[0,0], traj[0,1]))

        # 订阅控制指令
        self.cmd_sub = self.create_subscription(
            VehicleCmd, '/vehicle_cmd', self.cmd_callback, 10)

        # 发布车辆状态（20Hz）
        self.state_pub = self.create_publisher(VehicleState, '/vehicle_state', 10)
        self.timer = self.create_timer(0.05, self.publish_state)

    def cmd_callback(self, msg: VehicleCmd):
        """收到控制指令，推进仿真一步"""
        self.model.update(delta=msg.delta, a=msg.accel)

    def publish_state(self):
        """发布当前仿真状态"""
        state = self.model.get_state()  # [x, y, psi, v]

        msg = VehicleState()
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.x     = float(state[0])
        msg.y     = float(state[1])
        msg.theta = float(state[2])
        msg.v     = float(state[3])

        self.state_pub.publish(msg)


def main():
    rclpy.init()
    node = SimBridgeNode()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
```

## 第三阶段总结

完成本阶段后，你应该拥有：

| 产出 | 验证方式 |
|------|----------|
| ROS2 Humble 安装正常 | `ros2 --version` 输出正确 |
| 自定义消息编译成功 | `ros2 interface show av_control_msgs/msg/VehicleCmd` |
| 控制器节点运行 | `ros2 topic echo /vehicle_cmd` 有数据 |
| 轨迹发布节点运行 | `ros2 topic hz /trajectory` 约 1Hz |
| CARLA 桥接节点运行 | `ros2 topic hz /vehicle_state` 约 20Hz |
| launch 文件一键启动 | 一条命令启动全部节点 |
| CARLA 中车辆跟踪轨迹 | 目视验证 |

**常见错误：**
- `package not found` → 检查 `source install/setup.bash` 是否执行
- 节点启动后无控制指令 → 检查话题名称是否一致（大小写敏感）
- CARLA 连接超时 → 确认 CARLA 服务器先于桥接节点启动
- colcon build 报 Eigen 找不到 → 在 CMakeLists 加 `find_package(Eigen3 REQUIRED)`
