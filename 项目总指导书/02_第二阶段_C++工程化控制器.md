**理解 CMakeLists.txt 的结构：**

- `target_link_libraries`：告诉链接器找哪些库
- `find_package`：自动寻找系统中安装的第三方库

## 第六周：实现 C++ 控制器

### 6.1 车辆模型头文件

创建 `include/bicycle_model.hpp`：

```cpp
#pragma once
#include <Eigen/Dense>
#include <cmath>

class BicycleModel {
public:
    // 状态向量：[x, y, theta, v]
    using State = Eigen::Vector4d;

    struct Params {
        double wheelbase = 2.7;    // 轴距 (m)
        double dt = 0.05;          // 时间步长 (s)
        double max_steer = 0.5236; // 最大转角 30° (rad)
        double max_speed = 20.0;   // 最大速度 (m/s)
        double max_accel = 3.0;    // 最大加速度 (m/s²)
    };

    explicit BicycleModel(const Params& params = Params{});

    // 更新状态，返回新状态
    State update(const State& state, double delta, double accel) const;

    // 获取参数
    const Params& params() const { return params_; }

private:
    Params params_;
    static double normalizeAngle(double angle);
};
```

创建 `src/bicycle_model.cpp`：

```cpp
#include "bicycle_model.hpp"
#include <algorithm>

BicycleModel::BicycleModel(const Params& params) : params_(params) {}

BicycleModel::State BicycleModel::update(
    const State& state, double delta, double accel) const
{
    // 限幅
    delta = std::clamp(delta, -params_.max_steer, params_.max_steer);
    accel = std::clamp(accel, -params_.max_accel, params_.max_accel);

    const double x     = state[0];
    const double y     = state[1];
    const double theta = state[2];
    const double v     = state[3];
    const double dt    = params_.dt;

    // 欧拉积分
    State next;
    next[0] = x + v * std::cos(theta) * dt;
    next[1] = y + v * std::sin(theta) * dt;
    next[2] = normalizeAngle(theta + v / params_.wheelbase * std::tan(delta) * dt);
    next[3] = std::clamp(v + accel * dt, 0.0, params_.max_speed);

    return next;
}

double BicycleModel::normalizeAngle(double angle) {
    while (angle >  M_PI) angle -= 2.0 * M_PI;
    while (angle < -M_PI) angle += 2.0 * M_PI;
    return angle;
}
```

### 6.2 PID 控制器

创建 `include/pid_controller.hpp`：

```cpp
#pragma once

class PIDController {
public:
    PIDController(double kp, double ki, double kd, double dt = 0.05);

    double compute(double target, double current);
    void reset();

private:
    double kp_, ki_, kd_, dt_;
    double error_sum_;
    double last_error_;
};
```

创建 `src/pid_controller.cpp`：

```cpp
#include "pid_controller.hpp"
#include <algorithm>

PIDController::PIDController(double kp, double ki, double kd, double dt)
    : kp_(kp), ki_(ki), kd_(kd), dt_(dt),
      error_sum_(0.0), last_error_(0.0) {}

double PIDController::compute(double target, double current) {
    double error = target - current;

    // 积分项（加积分限幅防止积分饱和）
    error_sum_ += error * dt_;
    error_sum_ = std::clamp(error_sum_, -10.0, 10.0);

    double error_diff = (error - last_error_) / dt_;
    last_error_ = error;

    return kp_ * error + ki_ * error_sum_ + kd_ * error_diff;
}

void PIDController::reset() {
    error_sum_ = 0.0;
    last_error_ = 0.0;
}
```

### 6.3 Pure Pursuit 控制器

创建 `include/pure_pursuit.hpp`：

```cpp
#pragma once
#include <Eigen/Dense>
#include <vector>

class PurePursuitController {
public:
    struct Waypoint {
        double x, y, yaw, v_ref;
    };

    PurePursuitController(double wheelbase = 2.7,
                          double k = 0.1,
                          double ld_min = 2.0);

    // 返回 {转角(rad), 目标速度(m/s)}
    std::pair<double, double> compute(
        const Eigen::Vector4d& state,
        const std::vector<Waypoint>& trajectory);

private:
    double L_;
    double k_;
    double ld_min_;

    int findNearestIndex(const Eigen::Vector2d& pos,
                         const std::vector<Waypoint>& traj);
    static double normalizeAngle(double angle);
};
```

创建 `src/pure_pursuit.cpp`：

```cpp
#include "pure_pursuit.hpp"
#include <cmath>
#include <limits>
#include <algorithm>

PurePursuitController::PurePursuitController(
    double wheelbase, double k, double ld_min)
    : L_(wheelbase), k_(k), ld_min_(ld_min) {}

std::pair<double, double> PurePursuitController::compute(
    const Eigen::Vector4d& state,
    const std::vector<Waypoint>& trajectory)
{
    const double x = state[0], y = state[1];
    const double theta = state[2], v = state[3];

    // 前视距离随速度自适应
    double ld = std::max(k_ * v + ld_min_, ld_min_);

    // 找最近点
    Eigen::Vector2d pos(x, y);
    int nearest = findNearestIndex(pos, trajectory);

    // 从最近点向前找前视点
    int target = nearest;
    for (int i = nearest; i < (int)trajectory.size(); ++i) {
        double dist = std::hypot(trajectory[i].x - x, trajectory[i].y - y);
        if (dist >= ld) {
            target = i;
            break;
        }
    }

    // 计算前视点方向角
    double alpha = std::atan2(trajectory[target].y - y,
                               trajectory[target].x - x) - theta;
    alpha = normalizeAngle(alpha);

    // Pure Pursuit 公式
    double delta = std::atan2(2.0 * L_ * std::sin(alpha), ld);

    return {delta, trajectory[target].v_ref};
}

int PurePursuitController::findNearestIndex(
    const Eigen::Vector2d& pos, const std::vector<Waypoint>& traj)
{
    double min_dist = std::numeric_limits<double>::max();
    int idx = 0;
    for (int i = 0; i < (int)traj.size(); ++i) {
        double d = std::hypot(traj[i].x - pos[0], traj[i].y - pos[1]);
        if (d < min_dist) {
            min_dist = d;
            idx = i;
        }
    }
    return idx;
}

double PurePursuitController::normalizeAngle(double angle) {
    while (angle >  M_PI) angle -= 2.0 * M_PI;
    while (angle < -M_PI) angle += 2.0 * M_PI;
    return angle;
}
```

### 6.4 Stanley 控制器

**原理：** Stanley 方法是斯坦福大学 DARPA 挑战赛冠军车使用的控制器。与 Pure Pursuit 不同，它同时考虑**航向误差**和**横向误差**。

```
转角公式：δ = θ_e + arctan(k · e_y / v)

其中：
θ_e = 轨迹切线方向 - 当前航向角（航向误差）
e_y = 前轴到最近轨迹点的横向距离（横向位置误差）
k   = 增益系数
v   = 当前速度（速度大时减小横向修正，防止过激）
```

创建 `include/stanley_controller.hpp`：

```cpp
#pragma once
#include <Eigen/Dense>
#include <vector>

class StanleyController {
public:
    struct Waypoint {
        double x, y, yaw, v_ref;
    };

    StanleyController(double wheelbase = 2.7, double k = 0.5, double ks = 1.0);

    std::pair<double, double> compute(
        const Eigen::Vector4d& state,
        const std::vector<Waypoint>& trajectory);

private:
    double L_;
    double k_;   // 横向误差增益
    double ks_;  // 软化系数（防止低速分母为零）
    static double normalizeAngle(double angle);
};
```

创建 `src/stanley_controller.cpp`：

```cpp
#include "stanley_controller.hpp"
#include <cmath>
#include <limits>

StanleyController::StanleyController(double wheelbase, double k, double ks)
    : L_(wheelbase), k_(k), ks_(ks) {}

std::pair<double, double> StanleyController::compute(
    const Eigen::Vector4d& state,
    const std::vector<Waypoint>& trajectory)
{
    const double x = state[0], y = state[1];
    const double theta = state[2], v = state[3];

    // 计算前轴位置
    double front_x = x + L_ * std::cos(theta);
    double front_y = y + L_ * std::sin(theta);

    // 找前轴最近的轨迹点
    double min_dist = std::numeric_limits<double>::max();
    int nearest = 0;
    for (int i = 0; i < (int)trajectory.size(); ++i) {
        double d = std::hypot(trajectory[i].x - front_x,
                              trajectory[i].y - front_y);
        if (d < min_dist) {
            min_dist = d;
            nearest = i;
        }
    }

    // 航向误差
    double theta_ref = trajectory[nearest].yaw;
    double theta_e = normalizeAngle(theta_ref - theta);

    // 横向误差（带符号）
    double dx = front_x - trajectory[nearest].x;
    double dy = front_y - trajectory[nearest].y;
    // 投影到轨迹法线方向
    double e_y = -dx * std::sin(theta_ref) + dy * std::cos(theta_ref);

    // Stanley 公式
    double delta = theta_e + std::atan2(k_ * e_y, ks_ + v);

    return {delta, trajectory[nearest].v_ref};
}

double StanleyController::normalizeAngle(double angle) {
    while (angle >  M_PI) angle -= 2.0 * M_PI;
    while (angle < -M_PI) angle += 2.0 * M_PI;
    return angle;
}
```

### 7.3 主程序（C++ 独立仿真，输出 CSV）

创建 `src/main.cpp`：

```cpp
#include <iostream>
#include <fstream>
#include <vector>
#include <cmath>
#include "bicycle_model.hpp"
#include "pid_controller.hpp"
#include "pure_pursuit.hpp"
#include "stanley_controller.hpp"

// 生成 S 形轨迹
std::vector<PurePursuitController::Waypoint> generateSCurve() {
    std::vector<PurePursuitController::Waypoint> traj;
    const int N = 300;
    const double length = 100.0;
    for (int i = 0; i < N; ++i) {
        double x = length * i / N;
        double y = 10.0 * std::sin(2.0 * M_PI * x / length);
        double dx = length / N;
        double dy = 10.0 * std::cos(2.0 * M_PI * x / length) * (2.0 * M_PI / length) * dx;
        double yaw = std::atan2(dy, dx);
        traj.push_back({x, y, yaw, 6.0});
    }
    return traj;
}

int main() {
    auto traj = generateSCurve();

    BicycleModel model;
    PIDController pid(1.5, 0.1, 0.05);
    PurePursuitController pp;

    // 初始状态
    BicycleModel::State state;
    state << traj[0].x, traj[0].y, traj[0].yaw, 3.0;

    std::ofstream csv("results/simulation_output.csv");
    csv << "step,x,y,theta,v,lateral_error\n";

    double total_sq_error = 0.0;
    const int STEPS = 800;

    for (int step = 0; step < STEPS; ++step) {
        auto [delta, v_ref] = pp.compute(state, traj);
        double accel = pid.compute(v_ref, state[3]);
        state = model.update(state, delta, accel);

        // 计算横向误差
        double min_dist = 1e9;
        for (auto& wp : traj) {
            double d = std::hypot(wp.x - state[0], wp.y - state[1]);
            min_dist = std::min(min_dist, d);
        }
        total_sq_error += min_dist * min_dist;

        csv << step << "," << state[0] << "," << state[1] << ","
            << state[2] << "," << state[3] << "," << min_dist << "\n";
    }

    double rmse = std::sqrt(total_sq_error / STEPS);
    std::cout << "Pure Pursuit RMSE: " << rmse << " m\n";

    return 0;
}
```

## 第八周：C++ 对接 CARLA Python 桥接

C++ 控制器通过**标准输入输出（stdin/stdout）管道**与 Python CARLA 脚本通信。这是一种轻量级的进程间通信方式，不需要 ROS2 也能联调。

**数据流：**

```
CARLA Python 脚本
  → 通过 stdin 发送车辆状态给 C++ 程序
  ← 通过 stdout 接收控制指令
  → 发送给 CARLA 车辆
```

创建 `src/carla_interface.cpp`（替换 main.cpp 作为 CARLA 模式）：

```cpp
#include <iostream>
#include <fstream>
#include <string>
#include <sstream>
#include "bicycle_model.hpp"
#include "pid_controller.hpp"
#include "pure_pursuit.hpp"
#include <vector>

// 从文件加载轨迹（Python 预先生成）
std::vector<PurePursuitController::Waypoint> loadTrajectory(const std::string& path) {
    std::vector<PurePursuitController::Waypoint> traj;
    std::ifstream f(path);
    std::string line;
    std::getline(f, line); // 跳过表头
    while (std::getline(f, line)) {
        std::istringstream ss(line);
        PurePursuitController::Waypoint wp;
        char comma;
        ss >> wp.x >> comma >> wp.y >> comma >> wp.yaw >> comma >> wp.v_ref;
        traj.push_back(wp);
    }
    return traj;
}

int main() {
    auto traj = loadTrajectory("trajectory.csv");
    PIDController pid(1.5, 0.1, 0.05);
    PurePursuitController pp;

    double x, y, theta, v;
    // 从 stdin 持续读取车辆状态，输出控制指令
    while (std::cin >> x >> y >> theta >> v) {
        Eigen::Vector4d state(x, y, theta, v);
        auto [delta, v_ref] = pp.compute(state, traj);
        double accel = pid.compute(v_ref, v);
        double throttle = std::max(accel / 3.0, 0.0);
        double brake    = std::max(-accel / 5.0, 0.0);
        double steer    = delta / 1.2217;  // 1.2217 rad = 70°，与 carla_env.py 保持一致
        std::cout << throttle << " " << steer << " " << brake << std::endl;
    }
    return 0;
}
```

**常见错误：**

- 链接错误 `undefined reference` → 检查 CMakeLists.txt 的 `target_link_libraries`
- CARLA 桥接收不到数据 → 检查 Python 端的 `flush=True` 是否忘写
